# cbar Implementation Plan

> **For agentic workers:** REQUIRED SUB-SKILL: Use superpowers:subagent-driven-development (recommended) or superpowers:executing-plans to implement this plan task-by-task. Steps use checkbox (`- [ ]`) syntax for tracking.

**Goal:** A native macOS menu-bar app (`Cbar.app`) that shows every cswap-managed Claude account's usage as a color-coded icon + SwiftUI popover with progress bars, and switches accounts — replacing the `cswap-bar` SwiftBar plugin.

**Architecture:** A `CbarCore` library holds pure, testable logic (provider-generic `Account`/`Meter` model, cswap JSON decode, health-color rules). The `Cbar` executable is an AppKit menu-bar accessory (`NSStatusItem` + `NSPopover` hosting SwiftUI) driven by a polling `UsageStore`. A `CbarSelfTest` executable asserts the core logic (no XCTest, since only Command Line Tools are installed).

**Tech Stack:** Swift 6.3, SwiftPM, AppKit, SwiftUI, Foundation. No external dependencies. No Xcode (build via `swift build` + a script that assembles the `.app` bundle by hand).

## Global Constraints

- **Toolchain:** Swift 6.3 via Command Line Tools only — **no full Xcode**. No `xcodebuild`, no XCTest. Tests are `assert`-based executables run with `swift run`.
- **Dependencies:** none beyond system frameworks (Foundation, AppKit, SwiftUI). Never add a SwiftPM dependency.
- **Platform target:** `.macOS(.v13)` (dev machine is macOS 26).
- **cswap path:** `~/.local/bin/cswap` (i.e. `\(NSHomeDirectory())/.local/bin/cswap`).
- **Bundle:** name `Cbar`, display name `cbar`, id `com.heymo.cbar`, `LSUIElement = true`.
- **Health thresholds:** `status != "ok"` → crit; else `pct > 85` → crit, `pct >= 60` → warn, else healthy.
- **Menu-bar icon color = worst account** across all accounts (owner's decision). Stale (any account `ageSeconds > 600`) → icon dimmed to 50% alpha.
- **Poll interval:** 30 s, plus refresh on popover open and after any switch.
- **Fable meter** = the `scoped` entry `name == "Fable"`, displayed with id `"Fbl"`.

```swift
import Foundation

public struct Meter: Identifiable, Sendable {
    public let id: String          // "5h" | "7d" | "Fbl" | other scoped name
    public let pct: Double
    public let countdown: String?
    public init(id: String, pct: Double, countdown: String?) {
        self.id = id; self.pct = pct; self.countdown = countdown
    }
}

public struct Account: Identifiable, Sendable {
    public let id: String          // "cswap:2"
    public let number: Int
    public let email: String
    public let org: String
    public let isActive: Bool
    public let status: String      // "ok" | other
    public let meters: [Meter]
    public let ageSeconds: Double?
    public var maxPct: Double { meters.map(\.pct).max() ?? 0 }
    public init(id: String, number: Int, email: String, org: String,
                isActive: Bool, status: String, meters: [Meter], ageSeconds: Double?) {
        self.id = id; self.number = number; self.email = email; self.org = org
        self.isActive = isActive; self.status = status; self.meters = meters; self.ageSeconds = ageSeconds
    }
}

public enum Health: Sendable, Equatable { case healthy, warn, crit }

public func healthLevel(pct: Double, status: String) -> Health {
    if status != "ok" { return .crit }
    if pct > 85 { return .crit }
    if pct >= 60 { return .warn }
    return .healthy
}

public func overallHealth(_ accounts: [Account]) -> Health {
    var worst = Health.healthy
    for a in accounts {
        switch healthLevel(pct: a.maxPct, status: a.status) {
        case .crit: return .crit
        case .warn: worst = .warn
        case .healthy: break
        }
    }
    return worst
}

public func anyStale(_ accounts: [Account], threshold: Double = 600) -> Bool {
    accounts.contains { ($0.ageSeconds ?? 0) > threshold }
}

public protocol Provider {
    var name: String { get }
    func accounts() throws -> [Account]
    func switchTo(_ account: Account) throws
    func switchToBest() throws
}
```

Create `Sources/CbarCore/CswapClient.swift`:

```swift
import Foundation

public struct CswapClient: Provider {
    public let name = "cswap"
    private let path: String

    public init(path: String = "\(NSHomeDirectory())/.local/bin/cswap") {
        self.path = path
    }

    public func accounts() throws -> [Account] {
        try Self.decode(try run(["--list", "--json"]))
    }

    public func switchTo(_ account: Account) throws {
        _ = try run(["--switch-to", "\(account.number)"])
    }

    public func switchToBest() throws {
        _ = try run(["--switch", "--strategy", "best"])
    }

    @discardableResult
    private func run(_ args: [String]) throws -> Data {
        let p = Process()
        p.executableURL = URL(fileURLWithPath: path)
        p.arguments = args
        let out = Pipe()
        p.standardOutput = out
        p.standardError = Pipe()
        try p.run()
        let data = out.fileHandleForReading.readDataToEndOfFile()
        p.waitUntilExit()
        if p.terminationStatus != 0 { throw CswapError.exit(Int(p.terminationStatus)) }
        return data
    }

    public static func decode(_ data: Data) throws -> [Account] {
        let root = try JSONDecoder().decode(Root.self, from: data)
        return root.accounts.map { r in
            var meters: [Meter] = []
            if let f = r.usage?.fiveHour { meters.append(Meter(id: "5h", pct: f.pct ?? 0, countdown: f.countdown)) }
            if let s = r.usage?.sevenDay { meters.append(Meter(id: "7d", pct: s.pct ?? 0, countdown: s.countdown)) }
            for sc in r.usage?.scoped ?? [] {
                let id = sc.name == "Fable" ? "Fbl" : sc.name
                meters.append(Meter(id: id, pct: sc.pct ?? 0, countdown: sc.countdown))
            }
            return Account(id: "cswap:\(r.number)", number: r.number, email: r.email,
                           org: r.organizationName ?? "", isActive: r.active,
                           status: r.usageStatus ?? "ok", meters: meters, ageSeconds: r.usageAgeSeconds)
        }
    }

    enum CswapError: Error { case exit(Int) }

    private struct Root: Decodable { let accounts: [Raw] }
    private struct Raw: Decodable {
        let number: Int; let email: String; let organizationName: String?
        let active: Bool; let usageStatus: String?; let usage: RawUsage?; let usageAgeSeconds: Double?
    }
    private struct RawUsage: Decodable { let fiveHour: RawWin?; let sevenDay: RawWin?; let scoped: [RawScoped]? }
    private struct RawWin: Decodable { let pct: Double?; let countdown: String? }
    private struct RawScoped: Decodable { let pct: Double?; let countdown: String?; let name: String }
}
```

```swift
import AppKit
import CbarCore

final class StatusItemController: NSObject {
    private let item: NSStatusItem
    private let popover = NSPopover()

    init(popoverContent: NSViewController?) {
        item = NSStatusBar.system.statusItem(withLength: NSStatusItem.variableLength)
        super.init()
        if let content = popoverContent {
            popover.behavior = .transient
            popover.contentViewController = content
        }
        item.button?.target = self
        item.button?.action = #selector(togglePopover)
        render(overall: .healthy, stale: false, tooltip: "cbar")
    }

    var onPopoverOpen: (() -> Void)?

    @objc private func togglePopover() {
        guard let button = item.button, popover.contentViewController != nil else { return }
        if popover.isShown {
            popover.performClose(nil)
        } else {
            onPopoverOpen?()
            popover.show(relativeTo: button.bounds, of: button, preferredEdge: .minY)
            popover.contentViewController?.view.window?.makeKey()
        }
    }

    func render(overall: Health, stale: Bool, tooltip: String) {
        var color = overall.nsColor
        if stale { color = color.withAlphaComponent(0.5) }
        let cfg = NSImage.SymbolConfiguration(pointSize: 15, weight: .regular)
            .applying(.init(paletteColors: [color]))
        let img = NSImage(systemSymbolName: "arrow.left.arrow.right.circle.fill",
                          accessibilityDescription: "cbar")?
            .withSymbolConfiguration(cfg)
        img?.isTemplate = false
        item.button?.image = img
        item.button?.toolTip = tooltip
    }
}
```

```swift
import Foundation
import CbarCore

@Observable
final class UsageStore {
    private(set) var accounts: [Account] = []
    private(set) var lastError: String?
    private(set) var lastUpdated: Date?

    var onUpdate: (() -> Void)?

    private let provider: Provider
    private var timer: Timer?
    private let interval: TimeInterval = 30   // Task 2: raise to 120 if --list is a network fetch

    init(provider: Provider = CswapClient()) { self.provider = provider }

    func start() {
        refresh()
        timer = Timer.scheduledTimer(withTimeInterval: interval, repeats: true) { [weak self] _ in
            self?.refresh()
        }
    }

    func refresh() {
        DispatchQueue.global(qos: .utility).async { [weak self] in
            guard let self else { return }
            do {
                let accs = try self.provider.accounts()
                DispatchQueue.main.async {
                    self.accounts = accs
                    self.lastError = nil
                    self.lastUpdated = Date()
                    self.onUpdate?()
                }
            } catch {
                DispatchQueue.main.async {
                    self.lastError = String(describing: error)
                    self.onUpdate?()
                }
            }
        }
    }

    func switchTo(_ account: Account) {
        DispatchQueue.global(qos: .userInitiated).async { [weak self] in
            try? self?.provider.switchTo(account)
            self?.refresh()
        }
    }

    func switchToBest() {
        DispatchQueue.global(qos: .userInitiated).async { [weak self] in
            try? self?.provider.switchToBest()
            self?.refresh()
        }
    }

    var cacheAgeText: String {
        guard let age = accounts.compactMap(\.ageSeconds).max() else { return "—" }
        return "cached \(Int(age))s ago"
    }
}
```

```swift
import AppKit
import SwiftUI
import CbarCore

final class AppDelegate: NSObject, NSApplicationDelegate {
    private var store: UsageStore!
    private var statusCtl: StatusItemController!

    func applicationDidFinishLaunching(_ notification: Notification) {
        store = UsageStore()
        let host = NSHostingController(rootView: PopoverView(store: store))
        statusCtl = StatusItemController(popoverContent: host)
        statusCtl.onPopoverOpen = { [weak self] in self?.store.refresh() }
        store.onUpdate = { [weak self] in
            guard let self else { return }
            let overall = overallHealth(self.store.accounts)
            let stale = anyStale(self.store.accounts)
            let active = self.store.accounts.first(where: \.isActive)
            let tip = self.store.lastError
                ?? active.map { "\($0.email) · \(Int($0.maxPct.rounded()))%" }
                ?? "cbar"
            self.statusCtl.render(overall: overall, stale: stale, tooltip: tip)
        }
        store.start()
    }
}
```

```bash
#!/bin/bash
set -euo pipefail
cd "$(dirname "$0")"

./build.sh

PLIST="$HOME/Library/LaunchAgents/com.heymo.cbar.plist"
cat > "$PLIST" <<PL
<?xml version="1.0" encoding="UTF-8"?>
<!DOCTYPE plist PUBLIC "-//Apple//DTD PLIST 1.0//EN" "http://www.apple.com/DTDs/PropertyList-1.0.dtd">
<plist version="1.0">
<dict>
  <key>Label</key><string>com.heymo.cbar</string>
  <key>ProgramArguments</key>
  <array><string>$DEST/Contents/MacOS/Cbar</string></array>
  <key>RunAtLoad</key><true/>
</dict>
</plist>
PL

launchctl unload "$PLIST" 2>/dev/null || true
launchctl load "$PLIST"
echo "installed to $DEST and loaded LaunchAgent (starts at login)"
```
